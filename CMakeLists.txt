cmake_minimum_required(VERSION 3.24)
project(archbench VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ARCHBENCH_ENABLE_CUDA "Build GPU benchmark workers" ON)
option(ARCHBENCH_ALLOW_OLDER_CUDA "Allow unsupported older toolkit for development" OFF)
option(ARCHBENCH_ENABLE_FFT "Build cuFFT benchmarks" ON)
option(ARCHBENCH_ENABLE_SPARSE "Build cuSPARSE benchmarks" ON)
option(ARCHBENCH_COMPILE_CAPABILITY_PROBES "Compile exact optional MMA instructions before enabling their native request paths" ON)
set(ARCHBENCH_CUDA_ARCHS "80;89;120a" CACHE STRING "Exact architecture worker targets")

include(CTest)
add_executable(codec_host tests/codec_host.cpp)
target_include_directories(codec_host PRIVATE include)
target_compile_options(codec_host PRIVATE "$<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Werror;-UNDEBUG>")
add_test(NAME codec_exhaustive_host COMMAND codec_host)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME limb_reference_host COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_alu_reference.py --compiler ${CMAKE_CXX_COMPILER})
  add_test(NAME python_host_tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_*.py)
  set_tests_properties(python_host_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/python")
endif()

if(ARCHBENCH_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "CUDA compiler missing. Use CUDA 13.5 or set ARCHBENCH_ENABLE_CUDA=OFF for host tests.")
  endif()
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.5 AND NOT ARCHBENCH_ALLOW_OLDER_CUDA)
    message(FATAL_ERROR "This project targets CUDA 13.5+. Older builds require explicit ARCHBENCH_ALLOW_OLDER_CUDA=ON and do not establish CUDA13.5 compatibility.")
  endif()
  find_package(CUDAToolkit REQUIRED)
  if(NOT UNIX)
    message(FATAL_ERROR "Initial implementation targets Linux/WSL2 (POSIX process isolation and host __int128).")
  endif()
  file(GLOB_RECURSE fingerprint_inputs CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*" "${CMAKE_CURRENT_SOURCE_DIR}/include/*" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/*")
  list(APPEND fingerprint_inputs "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/tools/classify_mma_sass.py")
  list(SORT fingerprint_inputs)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${fingerprint_inputs})
  set(fingerprint_material "")
  foreach(input IN LISTS fingerprint_inputs)
    file(SHA256 "${input}" content_hash)
    file(RELATIVE_PATH relative_input "${CMAKE_CURRENT_SOURCE_DIR}" "${input}")
    string(APPEND fingerprint_material "${relative_input}:${content_hash}\n")
  endforeach()
  string(SHA256 source_sha256 "${fingerprint_material}")
  string(SHA256 flags_sha256 "${CMAKE_CUDA_FLAGS}|${CMAKE_CUDA_FLAGS_RELEASE}|${CMAKE_CXX_FLAGS}|${CMAKE_CXX_FLAGS_RELEASE}|${CMAKE_BUILD_TYPE}|O3_lineinfo_no_fast_math_no_rdc|${ARCHBENCH_ENABLE_FFT}|${ARCHBENCH_ENABLE_SPARSE}|${ARCHBENCH_COMPILE_CAPABILITY_PROBES}")
  include(cmake/CapabilityProbes.cmake)
  foreach(arch IN LISTS ARCHBENCH_CUDA_ARCHS)
    if(NOT arch MATCHES "^(80|89|120a)$")
      message(FATAL_ERROR "Supported exact targets: 80, 89, 120a (got ${arch})")
    endif()
    string(REGEX REPLACE "a$" "" numeric_arch "${arch}")
    set(worker archbench_sm${arch})
    add_executable(${worker} src/main.cu src/common.cu src/alu.cu src/gemm.cu src/systems.cu src/systems_software.cu src/quant.cu)
    target_include_directories(${worker} PRIVATE include)
    target_compile_features(${worker} PRIVATE cxx_std_17 cuda_std_17)
    set_target_properties(${worker} PROPERTIES CUDA_ARCHITECTURES OFF CUDA_SEPARABLE_COMPILATION OFF
      RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
    target_compile_options(${worker} PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo;--expt-relaxed-constexpr;-gencode=arch=compute_${arch},code=sm_${arch};-Xptxas=-v>")
    target_compile_definitions(${worker} PRIVATE ARCHBENCH_TARGET_SM=${numeric_arch}
      ARCHBENCH_TARGET_NAME="sm_${arch}" ARCHBENCH_VERSION="${PROJECT_VERSION}"
      ARCHBENCH_SOURCE_SHA256="${source_sha256}" ARCHBENCH_FLAGS_SHA256="${flags_sha256}"
      ARCHBENCH_NVCC_VERSION="${CMAKE_CUDA_COMPILER_VERSION}")
    archbench_probe_mma("${arch}" int4_mma has_int4_mma)
    archbench_probe_mma("${arch}" binary_mma has_binary_mma)
    archbench_probe_mma("${arch}" fp64_mma has_fp64_mma)
    target_compile_definitions(${worker} PRIVATE
      ARCHBENCH_INT4_MMA_CODEGEN="${has_int4_mma_CODEGEN}"
      ARCHBENCH_BINARY_MMA_CODEGEN="${has_binary_mma_CODEGEN}"
      ARCHBENCH_FP64_MMA_CODEGEN="${has_fp64_mma_CODEGEN}")
    if(arch STREQUAL "120a")
      target_compile_definitions(${worker} PRIVATE ARCHBENCH_ARCH_SUFFIX_A=1)
      if(has_int4_mma)
        target_compile_definitions(${worker} PRIVATE ARCHBENCH_SM120_INT4_MMA=1)
      endif()
      if(has_binary_mma)
        target_compile_definitions(${worker} PRIVATE ARCHBENCH_SM120_BINARY_MMA=1)
      endif()
    endif()
    if(has_fp64_mma)
      target_compile_definitions(${worker} PRIVATE ARCHBENCH_HAS_FP64_MMA=1)
    endif()
    target_link_libraries(${worker} PRIVATE CUDA::cudart CUDA::cublas CUDA::cublasLt)
    if(ARCHBENCH_ENABLE_FFT)
      target_link_libraries(${worker} PRIVATE CUDA::cufft)
      target_compile_definitions(${worker} PRIVATE ARCHBENCH_HAS_CUFFT=1)
    endif()
    if(ARCHBENCH_ENABLE_SPARSE)
      target_link_libraries(${worker} PRIVATE CUDA::cusparse)
      target_compile_definitions(${worker} PRIVATE ARCHBENCH_HAS_CUSPARSE=1)
    endif()
  endforeach()
endif()
